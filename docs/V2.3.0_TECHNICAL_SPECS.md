```python
from pyotp import TOTP, random_base32
import qrcode

class TwoFactorManager:
    """Manages 2FA authentication."""

    def generate_secret(self) -> str:
        """Generate new TOTP secret (base32)."""
        return random_base32()

    def generate_qr_code(
        self,
        username: str,
        secret: str,
        issuer: str = "WebScrape-TUI"
    ) -> bytes:
        """
        Generate QR code for authenticator app.

        Returns:
            PNG image bytes
        """
        totp = TOTP(secret)
        uri = totp.provisioning_uri(
            name=username,
            issuer_name=issuer
        )

        qr = qrcode.QRCode(version=1, box_size=10, border=5)
        qr.add_data(uri)
        qr.make(fit=True)

        img = qr.make_image(fill_color="black", back_color="white")

        # Convert to PNG bytes
        from io import BytesIO
        buffer = BytesIO()
        img.save(buffer, format='PNG')
        return buffer.getvalue()

    def verify_token(self, secret: str, token: str) -> bool:
        """
        Verify TOTP token.

        Args:
            secret: User's TOTP secret
            token: 6-digit code from authenticator app

        Returns:
            True if valid
        """
        totp = TOTP(secret)
        return totp.verify(token, valid_window=1)  # Allow 30s drift

    def enable_2fa(
        self,
        user_id: int,
        secret: str,
        verification_token: str
    ) -> Tuple[bool, str]:
        """
        Enable 2FA for user after verification.

        Returns:
            (success, message)
        """

    def disable_2fa(self, user_id: int) -> Tuple[bool, str]:
        """Disable 2FA for user."""

    def is_enabled(self, user_id: int) -> bool:
        """Check if 2FA is enabled for user."""
```

### 2.2 Backup Codes

```python
def generate_backup_codes(user_id: int, count: int = 10) -> List[str]:
    """
    Generate backup codes for 2FA recovery.

    Returns:
        List of 10 one-time use codes (8 chars each)
    """
    import secrets
    import string

    codes = []
    alphabet = string.ascii_uppercase + string.digits

    for _ in range(count):
        code = ''.join(secrets.choice(alphabet) for _ in range(8))
        # Format: XXXX-XXXX
        formatted = f"{code[:4]}-{code[4:]}"
        codes.append(formatted)

        # Store hashed version in database
        code_hash = hash_backup_code(formatted)
        store_backup_code(user_id, code_hash)

    return codes

def verify_backup_code(user_id: int, code: str) -> Tuple[bool, str]:
    """
    Verify and consume backup code.

    Returns:
        (valid, message)
    """
    # Hash provided code
    code_hash = hash_backup_code(code)

    # Check if exists and unused
    with get_db_connection() as conn:
        row = conn.execute("""
            SELECT id FROM backup_codes
            WHERE user_id = ? AND code_hash = ? AND used = 0
        """, (user_id, code_hash)).fetchone()

        if not row:
            return False, "Invalid or used backup code"

        # Mark as used
        conn.execute("""
            UPDATE backup_codes
            SET used = 1, used_at = ?
            WHERE id = ?
        """, (datetime.now(timezone.utc).isoformat(), row['id']))
        conn.commit()

    return True, "Backup code accepted"
```

### 2.3 Trusted Devices

```python
def trust_device(
    user_id: int,
    device_fingerprint: str,
    device_name: str,
    ip_address: str,
    user_agent: str,
    duration_days: int = 30
) -> int:
    """
    Mark device as trusted for 2FA.

    Returns:
        Trusted device ID
    """
    trusted_until = (
        datetime.now(timezone.utc) + timedelta(days=duration_days)
    ).isoformat()

    with get_db_connection() as conn:
        cursor = conn.execute("""
            INSERT INTO trusted_devices
            (user_id, device_fingerprint, device_name, ip_address,
             user_agent, trusted_until, created_at, last_used_at)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?)
        """, (
            user_id, device_fingerprint, device_name, ip_address,
            user_agent, trusted_until,
            datetime.now(timezone.utc).isoformat(),
            datetime.now(timezone.utc).isoformat()
        ))
        conn.commit()
        return cursor.lastrowid

def is_trusted_device(user_id: int, device_fingerprint: str) -> bool:
    """Check if device is trusted and not expired."""
    with get_db_connection() as conn:
        row = conn.execute("""
            SELECT trusted_until FROM trusted_devices
            WHERE user_id = ? AND device_fingerprint = ?
            AND trusted_until > ?
        """, (
            user_id,
            device_fingerprint,
            datetime.now(timezone.utc).isoformat()
        )).fetchone()

    return row is not None
```

### 2.4 Login Flow with 2FA

```python
def authenticate_with_2fa(
    username: str,
    password: str,
    totp_token: str = None,
    backup_code: str = None,
    device_fingerprint: str = None,
    trust_device: bool = False
) -> Tuple[Optional[int], Optional[str], str]:
    """
    Enhanced authentication with 2FA support.

    Args:
        username: Username
        password: Password
        totp_token: 6-digit TOTP code (if 2FA enabled)
        backup_code: Backup code (alternative to TOTP)
        device_fingerprint: Device fingerprint for trusted devices
        trust_device: Whether to mark device as trusted

    Returns:
        (user_id, session_token, message)
    """
    # Step 1: Validate username/password
    user_id, temp_session, msg = authenticate_user_enhanced(
        username, password, ip_address="127.0.0.1", user_agent="TUI"
    )

    if not user_id:
        return None, None, msg

    # Step 2: Check if 2FA enabled
    if not is_2fa_enabled(user_id):
        return user_id, temp_session, "Login successful"

    # Step 3: Check if device is trusted
    if device_fingerprint and is_trusted_device(user_id, device_fingerprint):
        update_device_last_used(user_id, device_fingerprint)
        return user_id, temp_session, "Login successful (trusted device)"

    # Step 4: Verify 2FA
    if totp_token:
        if verify_2fa_token(user_id, totp_token):
            if trust_device and device_fingerprint:
                trust_device(user_id, device_fingerprint, ...)
            return user_id, temp_session, "Login successful (2FA verified)"
        else:
            return None, None, "Invalid 2FA code"

    elif backup_code:
        valid, msg = verify_backup_code(user_id, backup_code)
        if valid:
            return user_id, temp_session, "Login successful (backup code used)"
        else:
            return None, None, msg

    else:
        # 2FA required but not provided
        return None, None, "2FA_REQUIRED"  # Special code for client
```

```python
        """, (user_id, password_hash, datetime.now(timezone.utc).isoformat()))

def check_password_reuse(
    user_id: int,
    new_password: str,
    history_count: int = 5
) -> bool:
    """
    Check if password was used recently.

    Returns:
        True if password is reused (should reject)
    """
    with get_db_connection() as conn:
        recent_hashes = conn.execute("""
            SELECT password_hash FROM password_history
            WHERE user_id = ?
            ORDER BY created_at DESC
            LIMIT ?
        """, (user_id, history_count)).fetchall()

    for row in recent_hashes:
        if bcrypt.checkpw(new_password.encode(), row['password_hash'].encode()):
            return True  # Password was used before

    return False  # Password is new

def cleanup_old_history(user_id: int, keep_count: int = 10) -> int:
    """Keep only last N passwords in history."""
    with get_db_connection() as conn:
        # Delete all except last N
        cursor = conn.execute("""
            DELETE FROM password_history
            WHERE user_id = ? AND id NOT IN (
                SELECT id FROM password_history
                WHERE user_id = ?
                ORDER BY created_at DESC
                LIMIT ?
            )
        """, (user_id, user_id, keep_count))
        conn.commit()
        return cursor.rowcount
```

### 3.2 Expiration Policy

```python
class PasswordExpirationPolicy:
    """Password expiration policy configuration."""

    def __init__(
        self,
        enabled: bool = True,
        expiration_days: int = 180,
        warning_days: int = 7,
        grace_days: int = 3,
        history_count: int = 5
    ):
        self.enabled = enabled
        self.expiration_days = expiration_days
        self.warning_days = warning_days
        self.grace_days = grace_days
        self.history_count = history_count

    @classmethod
    def from_env(cls) -> 'PasswordExpirationPolicy':
        """Load policy from environment."""
        return cls(
            enabled=os.getenv('ENABLE_PASSWORD_EXPIRATION', 'true').lower() == 'true',
            expiration_days=int(os.getenv('PASSWORD_EXPIRATION_DAYS', '180')),
            warning_days=int(os.getenv('PASSWORD_WARNING_DAYS', '7')),
            grace_days=int(os.getenv('PASSWORD_GRACE_DAYS', '3')),
            history_count=int(os.getenv('PASSWORD_HISTORY_COUNT', '5'))
        )

def check_password_expiration(user_id: int) -> Dict[str, Any]:
    """
    Check password expiration status.

    Returns:
        {
            'expired': bool,
            'expires_in_days': int,
            'warning': bool,
            'force_change': bool,
            'message': str
        }
    """
    policy = PasswordExpirationPolicy.from_env()

    if not policy.enabled:
        return {'expired': False, 'message': 'Expiration disabled'}

    with get_db_connection() as conn:
        user = conn.execute("""
            SELECT password_changed_at, force_password_change
            FROM users WHERE id = ?
        """, (user_id,)).fetchone()

    # Check force change flag
    if user['force_password_change']:
        return {
            'expired': True,
            'force_change': True,
            'message': 'Password change required by administrator'
        }

    # Check expiration
    if not user['password_changed_at']:
        # Never changed, force change
        return {
            'expired': True,
            'force_change': True,
            'message': 'Initial password change required'
        }

    changed_at = datetime.fromisoformat(user['password_changed_at'])
    now = datetime.now(timezone.utc)
    age_days = (now - changed_at).days

    expires_in = policy.expiration_days - age_days

    if age_days >= policy.expiration_days + policy.grace_days:
        return {
            'expired': True,
            'force_change': True,
            'message': f'Password expired {age_days - policy.expiration_days} days ago'
        }
    elif age_days >= policy.expiration_days:
        return {
            'expired': True,
            'force_change': False,
            'grace_days_left': policy.grace_days - (age_days - policy.expiration_days),
            'message': f'Password expired, {policy.grace_days - (age_days - policy.expiration_days)} grace days remaining'
        }
    elif expires_in <= policy.warning_days:
        return {
            'expired': False,
            'warning': True,
            'expires_in_days': expires_in,
            'message': f'Password expires in {expires_in} days'
        }
    else:
        return {
            'expired': False,
            'expires_in_days': expires_in,
            'message': 'Password valid'
        }
```

### 4.2 Alert Engine

```python
class SecurityAlertManager:
    """Manages security alerts."""

    def create_alert(
        self,
        user_id: int,
        alert_type: SecurityAlertType,
        severity: str,
        title: str,
        description: str,
        data: Dict[str, Any] = None,
        notify_email: bool = True,
        notify_inapp: bool = True
    ) -> int:
        """
        Create security alert.

        Args:
            user_id: User being alerted
            alert_type: Type of alert
            severity: low, medium, high, critical
            title: Alert title
            description: Detailed description
            data: Additional JSON data
            notify_email: Send email notification
            notify_inapp: Create in-app notification

        Returns:
            Alert ID
        """
        # Store alert
        alert_id = self._store_alert(
            user_id, alert_type, severity, title, description, data
        )

        # Send notifications based on user preferences
        if notify_inapp:
            NotificationManager().create_notification(
                user_id=user_id,
                type='security_alert',
                title=title,
                message=description,
                data={'alert_id': alert_id, 'severity': severity}
            )

        if notify_email:
            self._send_alert_email(user_id, title, description, severity)

        # Log to audit log
        get_audit_logger().log(
            event_type=AuditEventType.SECURITY_ALERT,
            user_id=user_id,
            event_data={'alert_id': alert_id, 'type': alert_type.value}
        )

        return alert_id
```

## 7. Performance Requirements

### 7.1 Email Queue Processing

- Email queue should process 100 emails/minute
- Failed emails retry with exponential backoff (1min, 5min, 15min, 1hr)
- Maximum 5 retry attempts before permanent failure
- Queue processing runs every 30 seconds

### 7.2 2FA Verification

- Token verification: <100ms
- QR code generation: <500ms
- Backup code verification: <50ms

### 7.3 Password Operations

- Password expiration check: <10ms
- Password history check: <50ms
- Expiration warning query: <100ms
